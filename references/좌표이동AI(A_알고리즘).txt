;---------------------- A* 및 기타 함수 ---------------------
AStar(map, start, goal) {
    openSet := {}
    cameFrom := {}

    gScore := {}
    fScore := {}

    openSet[start] := true
    gScore[start] := 0
    fScore[start] := Heuristic(start, goal)

    while (Count(openSet) > 0) {
        current := LowestFScore(openSet, fScore)
        if (current = goal)
            return ReconstructPath(cameFrom, current)

        openSet.Delete(current)
        x := StrSplit(current, ",")[1]
        y := StrSplit(current, ",")[2]

        for each, neighbor in GetNeighbors(map, x, y) {
            neighborKey := neighbor[1] "," neighbor[2]
            tentativeG := gScore[current] + 1

            if (!gScore.HasKey(neighborKey) || tentativeG < gScore[neighborKey]) {
                cameFrom[neighborKey] := current
                gScore[neighborKey] := tentativeG
                fScore[neighborKey] := tentativeG + Heuristic(neighborKey, goal)
                openSet[neighborKey] := true
            }
        }
    }
    return ""
}

Heuristic(a, b) {
    a := StrSplit(a, ",")
    b := StrSplit(b, ",")
    return Abs(a[1] - b[1]) + Abs(a[2] - b[2])
}

LowestFScore(set, fScore) {
    lowest := ""
    lowVal := 999999
    for key in set {
        if (fScore[key] < lowVal) {
            lowVal := fScore[key]
            lowest := key
        }
    }
    return lowest
}

GetNeighbors(map, x, y) {
    neighbors := []
    dirs := [[-1,0], [1,0], [0,-1], [0,1]] ; 좌우상하
    for each, dir in dirs {
        nx := x + dir[1]
        ny := y + dir[2]
        if (ny >= 1 && ny <= map.MaxIndex() && nx >= 1 && nx <= map[1].MaxIndex()) {
            if (map[ny][nx] = 0)
                neighbors.push([nx, ny])
        }
    }
    return neighbors
}

Count(set) {
    c := 0
    for _ in set
        c++
    return c
}

ReconstructPath(cameFrom, current) {
    path := [current]
    while (cameFrom.HasKey(current)) {
        current := cameFrom[current]
        path.InsertAt(1, current)
    }
    return Join(path)
}

Join(arr) {
    out := ""
    for i, v in arr {
        out .= v
        if (i < arr.MaxIndex())
            out .= " → "
    }
    return out
}

; === 함수: 화면에서 가장 가까운 좌표 찾기 ===
getClosestCoord(x1, y1, x2, y2, pattern, accX := 0.000001, accY := 0.000001, zoomX := 1, zoomY := 1)
{
    minDist := 999999
    closest := ""

    if (found := FindText(X, Y, x1, y1, x2, y2, accX, accY, pattern, 0,,,,,, zoomX, zoomY))
    {
        for index, item in found
        {
            dx := item.x
            dy := item.y
            pos := CalculateStatusCoordinates(dx, dy, xPos, yPos)

            dist := Sqrt((xPos - pos.fx)**2 + (yPos - pos.fy)**2)
            if (dist < minDist)
            {
                minDist := dist
                closest := {dx: dx, dy: dy, fx: pos.fx, fy: pos.fy}
            }
        }
    }
    return closest
}

; === 함수: 게임 격자 기준 좌표 계산 ===
CalculateStatusCoordinates(dx, dy, xPos, yPos)
{
    global cx, cy, GridSizeX, GridSizeY

    if (xPos <= 8) {
        fx := Round((dx - (cx - (8 * GridSizeX))) / GridSizeX)
    } else if (ix >= 9) {
        fx := xPos + Round((dx - cx) / GridSizeX)
    }

    if (yPos <= 7) {
        fy := Round((dy - (cy - (7 * GridSizeY))) / GridSizeY)
    } else if (iy >= 8) {
        fy := yPos + Round((dy - cy) / GridSizeY)
    }

    return {fx: fx, fy: fy}
}
